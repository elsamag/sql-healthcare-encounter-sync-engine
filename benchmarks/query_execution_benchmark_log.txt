    p.patient_id,
    p.first_name || ' ' || p.last_name AS patient_full_name,
    p.date_of_birth,
    p.gender,
    e.encounter_date,
    e.department_name,
    e.provider_id,
    e.diagnosis_code,
    e.encounter_type
FROM
    patients AS p
INNER JOIN
    clinical_encounters AS e
    ON p.patient_id = e.patient_id

    e.status = 'COMPLETED'

    e.encounter_date DESC,
    e.encounter_id ASC;
